Google Earth Engine (JavaScript) Mineral Prospecting Script 
// ============================================================
// Google Earth Engine — Zimbabwe National Mineral Intelligence Hub
// Sentinel-2 MSI Multi-Spectral Alteration & Geological Analysis
// ============================================================

var zimBoundary = ee.FeatureCollection("FAO/GAUL/2015/level0")
  .filter(ee.Filter.eq('ADM0_NAME', 'Zimbabwe'));

// Sentinel-2 Level-2A Surface Reflectance Harmonized
var s2 = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
  .filterBounds(zimBoundary)
  .filterDate('2025-01-01', '2026-03-31')
  .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 10));

// Mask clouds & shadows
function maskS2sr(image) {
  var qa = image.select('QA60');
  var cloudBitMask = 1 << 10;
  var cirrusBitMask = 1 << 11;
  var mask = qa.bitwiseAnd(cloudBitMask).eq(0)
    .and(qa.bitwiseAnd(cirrusBitMask).eq(0));
  return image.updateMask(mask).divide(10000);
}

var composite = s2.map(maskS2sr).median().clip(zimBoundary);

// 1. Iron Oxide Index (Gossan Indicator: B4/B2)
var ironOxide = composite.select('B4').divide(composite.select('B2'))
  .rename('iron_oxide');

// 2. Clay Minerals / Hydrothermal Alteration Index (B11/B12)
var clayIndex = composite.select('B11').divide(composite.select('B12'))
  .rename('clay_minerals');

// 3. Ferrous Iron Index (SWIR + NIR)
var ferrousIron = composite.select('B12').divide(composite.select('B8'))
  .rename('ferrous_iron');

// Visualization Palettes
Map.centerObject(zimBoundary, 7);
Map.addLayer(composite, {bands: ['B4', 'B3', 'B2'], min: 0, max: 0.3}, 'S2 True Color');
Map.addLayer(ironOxide, {min: 1.0, max: 2.8, palette: ['blue', 'cyan', 'yellow', 'orange', 'red']}, 'Iron Oxide Gossan (B4/B2)');
Map.addLayer(clayIndex, {min: 1.0, max: 2.2, palette: ['purple', 'blue', 'green', 'yellow']}, 'Hydrothermal Clay (B11/B12)');
